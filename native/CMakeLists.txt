cmake_minimum_required(VERSION 3.20)
project(siql_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(siql_native SHARED src/siql_native.cpp)
target_include_directories(siql_native PUBLIC include)
target_compile_definitions(siql_native PRIVATE SIQL_NATIVE_BUILD)
if(MSVC)
  target_compile_options(siql_native PRIVATE /W4 /EHsc)
else()
  target_compile_options(siql_native PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(siql_native_tests tests/native_tests.cpp)
target_link_libraries(siql_native_tests PRIVATE siql_native)
enable_testing()
add_test(NAME siql_native_tests COMMAND siql_native_tests)
